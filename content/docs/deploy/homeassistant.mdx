```yaml
services:
  homeassistant:
    image: ghcr.nju.edu.cn/home-assistant/home-assistant:stable
    container_name: homeassistant
    network_mode: bridge
    restart: unless-stopped
    volumes:
      - ./config:/config
    environment:
      - TZ=Asia/Shanghai
    ports:
      - "8123:8123"
```
