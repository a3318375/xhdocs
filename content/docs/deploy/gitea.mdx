`mkdir -p /data/gitea/{custom,data,log}`  
chown -R git:git /data/gitea/  
chmod -R 750 /data/gitea/  
mkdir /etc/gitea  
chown root:git /etc/gitea  
chmod 770 /etc/gitea

‍

​`GITEA_WORK_DIR=/data/gitea/ /usr/local/bin/gitea web -c /etc/gitea/app.ini`
